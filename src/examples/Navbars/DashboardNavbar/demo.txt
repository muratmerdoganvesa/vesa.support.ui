// ─── Pure Helpers ─────────────────────────────────────────────────────────────

const resolveMenuIconClass = (
  menuItem: any,
  parentMenuName?: string,
  siblingIndex = 0
): string => {
  const explicitIcon = menuItem?.icon;
  if (explicitIcon && explicitIcon !== "null" && explicitIcon !== "undefined") {
    return `pi pi-${explicitIcon}`;
  }

  const normalizedName = String(menuItem?.name || "").toLocaleLowerCase("tr-TR");
  const normalizedHref = String(menuItem?.href || "").toLocaleLowerCase("tr-TR");
  const composite = `${normalizedName} ${normalizedHref}`;
  const parentName = String(parentMenuName || "").toLocaleLowerCase("tr-TR");

  if (
    parentName.includes("yardim masasi") ||
    parentName.includes("yardım masası") ||
    parentName.includes("help desk")
  ) {
    if (composite.includes("taleplerim")) return "pi pi-inbox";
    if (composite.includes("talep yonetimi") || composite.includes("talep yönetimi"))
      return "pi pi-sitemap";
  }

  const matchedRule = MENU_ICON_RULES.find((r) =>
    r.keywords.some((kw) => composite.includes(kw))
  );
  if (matchedRule) return `pi pi-${matchedRule.icon}`;

  const matchedPool = PARENT_ICON_POOLS.find((p) =>
    p.parentKeywords.some((kw) => parentName.includes(kw))
  );
  if (matchedPool) {
    return `pi pi-${matchedPool.icons[siblingIndex % matchedPool.icons.length]}`;
  }

  return "pi pi-circle-fill";
};

// ─── Custom Hook ──────────────────────────────────────────────────────────────

const useIsDesktop = (breakpoint = 1024): boolean => {
  const [isDesktop, setIsDesktop] = useState(
    () => typeof window !== "undefined" && window.innerWidth >= breakpoint
  );

  useEffect(() => {
    const mq = window.matchMedia(`(min-width: ${breakpoint}px)`);
    const handler = (e: MediaQueryListEvent) => setIsDesktop(e.matches);
    mq.addEventListener("change", handler);
    return () => mq.removeEventListener("change", handler);
  }, [breakpoint]);

  return isDesktop;
};

// ─── DesktopMenuItem ──────────────────────────────────────────────────────────

interface DesktopMenuItemProps {
  item: any;
  pathname: string;
  parentIndex: number;
}

const DesktopMenuItem = ({ item, pathname, parentIndex }: DesktopMenuItemProps) => {
  const navigate = useNavigate();
  const hasSubMenus = Array.isArray(item?.subMenus) && item.subMenus.length > 0;

  const isActive = hasSubMenus
    ? (item.subMenus as any[]).some(
        (sub) => sub.href && sub.href !== "#" && pathname === sub.href
      )
    : !!(item.href && item.href !== "#" && pathname === item.href);

  const iconClass = resolveMenuIconClass(item, undefined, parentIndex);

  const triggerClass = cn(
    "inline-flex items-center gap-1.5 px-3 py-1.5 rounded-lg",
    "text-sm font-semibold whitespace-nowrap transition-all duration-150",
    "outline-none focus-visible:ring-2 focus-visible:ring-blue-400",
    isActive
      ? "bg-blue-600 text-white shadow-[0_3px_8px_rgba(37,99,235,0.22)]"
      : "text-blue-600 hover:bg-blue-50"
  );

  if (hasSubMenus) {
    return (
      <DropdownMenu>
        <DropdownMenuTrigger asChild>
          <button className={triggerClass} aria-current={isActive ? "page" : undefined}>
            <i className={cn(iconClass, "text-[0.85rem] leading-none")} aria-hidden="true" />
            {menuAPIController(item.name)}
            <ChevronDown className="h-3 w-3 opacity-60" aria-hidden="true" />
          </button>
        </DropdownMenuTrigger>
        <DropdownMenuContent
          align="start"
          className="min-w-[220px] rounded-xl border-slate-100 py-1.5 shadow-md shadow-slate-100/80"
        >
          {(item.subMenus as any[]).map((subItem, subIndex) => {
            const isSelected =
              subItem.href && subItem.href !== "#" && pathname === subItem.href;
            return (
              <DropdownMenuItem
                key={subItem.id}
                onClick={() => navigate(subItem.href || "#")}
                className={cn(
                  "mx-1 cursor-pointer gap-2.5 rounded-lg",
                  isSelected &&
                    "bg-blue-600 text-white focus:bg-blue-700 focus:text-white"
                )}
                aria-current={isSelected ? "page" : undefined}
              >
                <i
                  className={cn(
                    resolveMenuIconClass(subItem, item?.name, subIndex),
                    "text-sm shrink-0"
                  )}
                  aria-hidden="true"
                />
                {menuAPIController(subItem.name)}
              </DropdownMenuItem>
            );
          })}
        </DropdownMenuContent>
      </DropdownMenu>
    );
  }

  return (
    <button
      onClick={() => navigate(item.href || "#")}
      className={triggerClass}
      aria-current={isActive ? "page" : undefined}
    >
      <i className={cn(iconClass, "text-[0.85rem] leading-none")} aria-hidden="true" />

  const [isDeleteModalOpen, setIsDeleteModalOpen] = useState(false);
  const [loginMail, setLoginMail] = useState("");

  const [newPw, setNewPw] = useState("");
  const [newPwConfirm, setNewPwConfirm] = useState("");
  const [passwordError, setPasswordError] = useState("");
  const [pswTrue, setPswTrue] = useState(true);

  const { pathname } = useLocation();
  const navigate = useNavigate();
  const dispatchAlert = useAlert();
  const dispatchBusy = useBusy();
  const isDesktop = useIsDesktop();

  // Keep transparent navbar context in sync with scroll position
  useEffect(() => {
    const handleTransparentNavbar = () => {
      setTransparentNavbar(
        dispatch,
        (fixedNavbar && window.scrollY === 0) || !fixedNavbar
      );
    };
    window.addEventListener("scroll", handleTransparentNavbar);
    handleTransparentNavbar();
    return () => window.removeEventListener("scroll", handleTransparentNavbar);
  }, [dispatch, fixedNavbar]);

  // ── Queries ──────────────────────────────────────────────────────────────

  const { data: userDataQuery } = useQuery(
    "dashboardUserData",
    async () => {
      dispatchBusy({ isBusy: true });
      try {
        const conf = getConfiguration();
        const [ticketData, userDetail] = await Promise.all([
          new TicketApi(conf).apiTicketCheckPermGet(),
          new UserApi(conf).apiUserGetLoginUserDetailGet(),
        ]);
        return {
          ticketPermission: ticketData.data,
          userDetail: userDetail.data,
        };
      } finally {
        dispatchBusy({ isBusy: false });
      }
    },
    {
      staleTime: 1000 * 60 * 5,
      cacheTime: 1000 * 60 * 60,
      refetchOnMount: false,
      refetchOnWindowFocus: false,
      enabled: !!localStorage.getItem("accessToken"),
    }
  );

  const { data: topMenuItemsQuery } = useQuery(
    "menuItems",
    async () => {
      const conf = getConfigurationAccessTokenLogin();
      const data = await new MenuApi(conf).apiMenuGet();
      return data.data;
    },
    {
      staleTime: 1000 * 60 * 5,
      cacheTime: 1000 * 60 * 60,
      refetchOnWindowFocus: false,
      refetchOnMount: false,
      refetchOnReconnect: false,
      enabled: !!localStorage.getItem("accessToken"),
    }
  );

  // Keep menu data cached for other consumers
  useQuery(
    "dashboardMenuData",
    async () => {
      dispatchBusy({ isBusy: true });
      try {
        const conf = getConfiguration();
        const data = await new MenuApi(conf).apiMenuAllListDataGet();
        return data.data;
      } finally {
        dispatchBusy({ isBusy: false });
      }
    },
    {
      staleTime: 1000 * 60 * 5,
      cacheTime: 1000 * 60 * 60,
      refetchOnMount: false,
      refetchOnWindowFocus: false,
      enabled: !!localStorage.getItem("accessToken"),
    }
  );

  // ── Effects ───────────────────────────────────────────────────────────────

  useEffect(() => {
    if (!userDataQuery) return;
    setUserData(userDataQuery.ticketPermission);
    setLoginMail(userDataQuery.userDetail.email);
    getApproveDetail(userDataQuery.ticketPermission.id);
  }, [userDataQuery]);

  useEffect(() => {
    if (topMenuItemsQuery) {
      setTopMenuItems(topMenuItemsQuery as any[]);
    }
  }, [topMenuItemsQuery]);

  // ── Handlers ──────────────────────────────────────────────────────────────

  const getApproveDetail = async (_userId: any) => {
    const conf = getConfiguration();
    const result =
      await new ApproveItemsApi(conf).apiApproveItemsGetPendingCountGetPendingCountGet();
    setWaitingCount(result.data);
  };

  const handleChangePassword = () => setIsDeleteModalOpen(true);

  const handleDeleteCloseModal = () => {
    setNewPw("");
    setNewPwConfirm("");
    setPasswordError("");
    setPswTrue(true);
    setIsDeleteModalOpen(false);
  };

  const validatePassword = (password: string): string => {
    if (password.length < 6)
      return "Parola en az 6 karakter uzunluğunda olmalıdır";
    if (!/[!@#$%^&*(),.?":{}|<>]/.test(password))
      return "Parola en az bir özel karakter içermelidir";
    if (!/\d/.test(password)) return "Parola en az bir rakam içermelidir";
    if (!/[A-Z]/.test(password))
      return "Parola en az bir büyük harf (A-Z) içermelidir";
    return "";
  };

  const handlePasswordChange = (pw: string) => {
    setNewPw(pw);
    setPasswordError(validatePassword(pw));
  };

  const handleConfirmPasswordChange = (pw: string) => {
    setNewPwConfirm(pw);
    const error = validatePassword(pw);
    if (error) {
      setPasswordError(error);
      setPswTrue(true);
    } else if (newPw && pw !== newPw) {
      setPasswordError("Şifreler eşleşmiyor..!");
      setPswTrue(true);
    } else {
      setPasswordError("");
      setPswTrue(false);
    }
  };

  const changePassword = async () => {
    try {
      dispatchBusy({ isBusy: true });
      if (
        validatePassword(newPw) ||
        validatePassword(newPwConfirm) ||
        newPw !== newPwConfirm
      ) {
        dispatchAlert({
          message: "Şifre geçerli değil. Lütfen kontrol edin.",
          type: MessageBoxType.Error,
        });
        return;
      }
      const conf = getConfiguration();
      await new UserApi(conf).apiUserResetPassWordGet(loginMail, newPw);
      dispatchAlert({
        message: "Şifre başarıyla güncellendi",
        type: MessageBoxType.Success,
      });
      handleDeleteCloseModal();
    } catch (error) {
      dispatchAlert({ message: `Hata: ${error}`, type: MessageBoxType.Error });
      handleDeleteCloseModal();
    } finally {
      dispatchBusy({ isBusy: false });
    }
  };

  const handleNotificationClick = () => {
    if (waitingCount > 0) navigate("/approve");
  };

  // ── Render ────────────────────────────────────────────────────────────────

  return (
    <div className="sticky top-0 z-50 w-full">
      {/* ── Top Shell Bar ── */}
      <header
        className={cn(
          "flex items-center justify-between border-b px-4 py-2",
          "shadow-[0_1px_4px_rgba(15,23,42,0.06)]",
          darkMode
            ? "border-slate-700 bg-slate-900"
            : "border-slate-100 bg-white"
        )}
      >
        {/* Left: Logo + app title */}
        <div className="flex items-center gap-3">
          <img
            src={saplogo}
            alt="SAP Logo"
            className="h-7 w-auto"
            loading="lazy"
          />
          {userData?.name && (
            <span
              className={cn(
                "hidden text-sm font-semibold sm:block",
                darkMode ? "text-slate-200" : "text-slate-700"
              )}
            >
              {userData.name}
            </span>
          )}
        </div>

        {/* Right: Notifications + Profile */}
        <div className="flex items-center gap-1.5">
          <NotificationSection
            count={waitingCount}
            showEmpty={showNoNotification}
            onShowEmpty={setShowNoNotification}
            onGoToApprove={handleNotificationClick}
          />
          <ProfileMenu
            photo={userDataQuery?.userDetail?.photo}
            userName={userData?.name}
            onChangePassword={handleChangePassword}
          />
        </div>
      </header>

      {/* ── Menu Bar ── */}
      <nav
        className={cn(
          "border-b",
          darkMode
            ? "border-slate-700 bg-slate-900 shadow-[0_4px_12px_rgba(0,0,0,0.2)]"
            : "border-slate-100 bg-white shadow-[0_4px_12px_rgba(15,23,42,0.04)]"
        )}
        aria-label="Ana menü"
      >
        {isDesktop ? (
          <div
            className={cn(
              "flex gap-1 overflow-x-auto px-4 py-1.5",
              "[scrollbar-width:thin]",
              "[&::-webkit-scrollbar]:h-1",
              "[&::-webkit-scrollbar-thumb]:rounded-full",
              "[&::-webkit-scrollbar-thumb]:bg-slate-200"
            )}
          >
            {topMenuItems.map((item, idx) => (
              <DesktopMenuItem
                key={item.id}
                item={item}
                pathname={pathname}
                parentIndex={idx}
              />
            ))}
          </div>
        ) : (
          <div className="flex items-center justify-between px-4 py-2">
            <span
              className={cn(
                "text-sm font-semibold",
                darkMode ? "text-slate-300" : "text-slate-600"
              )}
            >
              {userData?.name ?? ""}
            </span>
            <Button
              variant="ghost"
              size="icon"
              onClick={() => setIsTopMenuDrawerOpen(true)}
              className="h-8 w-8"
              aria-label="Menüyü aç"
              aria-expanded={isTopMenuDrawerOpen}
            >
              <MenuIcon className="h-4 w-4" aria-hidden="true" />
            </Button>
          </div>
        )}
      </nav>

      {/* ── Mobile Sheet ── */}
      <MobileMenuSheet
        open={isTopMenuDrawerOpen}
        onClose={() => setIsTopMenuDrawerOpen(false)}
        items={topMenuItems}
        pathname={pathname}
        onNavigate={navigate}
      />

      {/* ── Password Dialog ── */}
      <ChangePasswordDialog
        open={isDeleteModalOpen}
        onClose={handleDeleteCloseModal}
        newPw={newPw}
        newPwConfirm={newPwConfirm}
        passwordError={passwordError}
        pswDisabled={pswTrue}
        onNewPwChange={handlePasswordChange}
        onConfirmPwChange={handleConfirmPasswordChange}
        onSubmit={changePassword}
      />
    </div>
  );
}

DashboardNavbar.defaultProps = {
  absolute: false,
  light: false,
  isMini: false,
};

export default DashboardNavbar;
